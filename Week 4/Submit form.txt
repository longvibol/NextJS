import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { usersTable } from "../../db/schema";
import { db } from "./../../db/drizzle";

async function createNewUser(formData: FormData) {
  "use server";

  const payload = {
    name: formData.get("name") as string,
    age: Number(formData.get("age")), // convert to number
    email: formData.get("email") as string,
  };

  console.log({ payload });

  await db
    .insert(usersTable)
    .values({ age: payload.age, email: payload.email, name: payload.name });
}

// get from server  
async function getAllUsers() {
  "use server"; // to clearify this is server function 
  return await db.select().from(usersTable)
}



export default async function Page() {

  // when it startup we mount from the server
  const users = await getAllUsers()

  return (
    <div className="max-w-7xl mx-auto p-4 grid grid-cols-2 gap-4">
      <form action={createNewUser}>
        <div className="max-w-2xl flex flex-col gap-4">
          <h1 className="text-2xl font-bold">Crate New User</h1>
          <Label>Name</Label>
          <Input name="name"></Input>

          <Label>Age</Label>
          <Input name="age"></Input>

          <Label>Email</Label>
          <Input name="email"></Input>

          <Button>Submit</Button>
        </div>
      </form>
      <section>
        User
        {/* {JSON.stringify(users, null, 2)} */}

        <ul>
          {users.map((user)=> (
            <li key={user.id}>
              Username: {user.name}, Age: {user.age}, Email: {user.email}
            </li>
          ))}
        </ul>
      </section>
    </div>
  );
}
